export function MetaTable({ rows }) {
  return (
    <table>
      <thead>
        <tr>
          <th>Name</th>
          <th>Tokens</th>
          <th>Notiz</th>
        </tr>
      </thead>
      <tbody>
        {rows.map((row) => (
          <tr key={row.name}>
            <td>
              <code>{row.name}</code>
            </td>
            <td>{row.tokens.length > 0 ? row.tokens.join(", ") : "—"}</td>
            <td>{row.note ?? ""}</td>
          </tr>
        ))}
      </tbody>
    </table>
  )
}
